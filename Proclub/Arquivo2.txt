using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        List<double> P1 = new List<double> { 7.0, 8.5, 6.0, 9.0, 7.5 };
        List<double> P2 = new List<double> { 8.0, 7.0, 7.5, 9.5, 8.5 };
        double somaP1 = 0;
        double somaP2 = 0;
        foreach (double nota in P1)
        {
            somaP1 += nota;
        }
        foreach (double nota in P2)
        {
            somaP2 += nota;
        }
        double mediaP1 = somaP1 / P1.Count;
        double mediaP2 = somaP2 / P2.Count;
        Console.WriteLine($"Média da Prova 1: {mediaP1:F2}");
        Console.WriteLine($"Média da Prova 2: {mediaP2:F2}");
        if (mediaP1 > mediaP2)
        {
            Console.WriteLine("A turma obteve a melhor média na Prova 1.");
        }
        else if (mediaP2 > mediaP1)
        {
            Console.WriteLine("A turma obteve a melhor média na Prova 2.");
        }
        else
        {
            Console.WriteLine("As duas provas tiveram a mesma média.");
        }
    }
}
